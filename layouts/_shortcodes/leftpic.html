{{- /* leftpic — float an image to the left of wrapping inner text, in a flex row. usage: {{% leftpic rsrc="pic.jpg" %}}text beside the picture{{% /leftpic %}} params (named): one of rsrc (page-resource glob) | asset (global asset) | url (raw URL) | nopic; width (px, default 200); gutter (px, default 10); alt; title; topmargin/bottommargin (default 1rem). Inner content is the right-hand text column. Errors if none of rsrc/asset/url/nopic is given. */ -}} {{- $topmargin := .Get "topmargin" | default "1rem" }} {{- $bottommargin := .Get "bottommargin" | default "1rem" }}
{{- $imgwidth := .Get "width" | default 200 }} {{- $gutter := .Get "gutter" | default 10 }} {{- $s := .Scratch }} {{- $p := .Page }} {{- $r := (.Get "rsrc") }} {{- $a := (.Get "asset") }} {{- $alt := .Get "alt" }}
{{- if (or $r $a (.Get "url") (.Get "nopic" ) ) }} {{- if (or $r (.Get "asset")) }} {{ with $r }} {{- $s.Set "bigimage" ($p.Resources.GetMatch $r) }} {{ end }} {{ with $a }} {{- $s.Set "bigimage" (resources.Get $a) }} {{ end }} {{- $size := $imgwidth | printf "%dx" }} {{- $bigimage := $s.Get "bigimage"}} {{- $image := $bigimage.Resize $size }} {{- $imageurl := $image.RelPermalink}} {{- $title := .Get "title" | default $image.Title }} {{- $alttxt := $alt | default $title | default "Left Picture" }} {{ else if .Get "url" }} {{- $imageurl := .Get "url" }} {{- $title := .Get "title" | default "Left Picture" }} {{- $alttxt := $alt | default $title | default "Left Picture" }} {{ end }} {{- else }} {{ errorf "leftpic needs an rsrc, asset or url (or nopic) at %s" .Position }} {{- end }}
