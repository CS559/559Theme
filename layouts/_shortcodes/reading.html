{{/* look up a reading and put the html in the page usage: {{< reading "reading-id" "assignment" >}} where "reading-id" is the key in data/readings.yaml and "type" is the kind of reading "required" "optional" "alternate" etc. */}} {{- $reading := .Get 0 -}} {{- $kind := .Get 1 -}} {{- $readinginfo := index hugo.Data.readings $reading -}} {{- with $readinginfo -}} {{- with $kind -}} {{- $type := . | lower -}} {{- if eq $type "required" -}} (required)  {{- else if eq $type "optional" -}} (optional)  {{- else if eq $type "alternate" -}} (alternate)  {{- else if eq $type "recommended" -}} (optional - but recommended)  {{- else if eq $type "summary" -}} (ok to read summary)  {{- else -}} ( {{ . }} )  {{- end -}} {{- end}} {{- $readinginfo | safeHTML -}} {{- else -}} {{- errorf "Reading '%s' not found in readings database (for %s at %s)" $reading .Page.File .Position -}} {{- end -}}
