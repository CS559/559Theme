{{/* look up a reading and put the html in the page usage: {{< reading "reading-id" "assignment" >}} where "reading-id" is the key in data/readings.yaml and "type" is the kind of reading "required" "optional" "alternate" etc. Each readings.yaml entry is a map: .html is the rendered citation, alongside the individual fields (title, cfile, doi, url, ...). */}} {{- $reading := .Get 0 -}} {{- $kind := .Get 1 -}} {{- $readinginfo := index hugo.Data.readings $reading -}} {{- if eq (printf "%T" $readinginfo) "string" -}} {{- errorf "data/readings.yaml is in the old plain-string format; regenerate it with themes/559Theme/tools/seatable-to-readings.py (reading '%s' for %s at %s)" $reading .Page.File .Position -}} {{- else -}} {{- with $readinginfo -}} {{- with $kind -}} {{- $type := . | lower -}} {{- if eq $type "required" -}} (required)  {{- else if eq $type "optional" -}} (optional)  {{- else if eq $type "alternate" -}} (alternate)  {{- else if eq $type "recommended" -}} (optional - but recommended)  {{- else if eq $type "summary" -}} (ok to read summary)  {{- else -}} ( {{ . }} )  {{- end -}} {{- end}} {{- .html | safeHTML -}} {{- else -}} {{- errorf "Reading '%s' not found in readings database (for %s at %s)" $reading .Page.File .Position -}} {{- end -}} {{- end -}}
